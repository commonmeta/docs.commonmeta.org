{{- /* Render the type of a JSON Schema node as HTML: $ref link, array of ..., or primitive. */ -}} {{- $p := . -}} {{- $s := "any" -}} {{- if reflect.IsMap $p -}} {{- with index $p "$ref" -}} {{- $name := path.Base . -}} {{- $s = printf `%s` (anchorize $name) $name -}} {{- else -}} {{- with $p.type -}} {{- if eq . "array" -}} {{- $s = printf "array of %s" (partial "schema/type.html" $p.items) -}} {{- else -}} {{- $s = printf "%s" . -}} {{- with $p.format }}{{ $s = printf "%s (%s)" $s . }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $s -}}
